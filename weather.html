const express = require('express'); const axios = require('axios'); const sqlite3 = require('sqlite3').verbose(); const crypto = require('crypto'); const app = express(); app.use(express.json()); // إعدادات البيئة (يتم ضبطها في السيرفر مثل Render) const BOT_TOKEN = process.env.BOT_TOKEN; const ADMIN_SECRET = process.env.ADMIN_SECRET; // قاعدة بيانات المستخدمين const db = new sqlite3.Database('./app_data.db'); db.run(`CREATE TABLE IF NOT EXISTS users ( id TEXT PRIMARY KEY, token TEXT, chatId TEXT, lat REAL, lon REAL, cityName TEXT, lastAlertState TEXT, lastQuakeId TEXT )`); // 1. تسجيل مستخدم جديد (محمي بكلمة سر) app.post('/api/register', (req, res) => { const { chatId, secret } = req.body; if (secret !== ADMIN_SECRET) return res.status(401).json({ message: "غير مصرح" }); const userId = crypto.randomUUID(); const token = crypto.randomBytes(16).toString('hex'); db.run("INSERT INTO users (id, token, chatId, lat, lon, cityName) VALUES (?,?,?,?,?,?)", [userId, token, chatId, 0, 0, "غير محدد"], (err) => { if (err) return res.status(500).json({ message: "خطأ في التسجيل" }); res.json({ userId, token }); }); }); // 2. تحديث الموقع (محمي بالتوكن) app.post('/api/update-location', (req, res) => { const { userId, token, lat, lon, cityName } = req.body; db.get("SELECT * FROM users WHERE id = ? AND token = ?", [userId, token], (err, user) => { if (!user) return res.status(403).json({ message: "توكن غير صالح" }); db.run("UPDATE users SET lat = ?, lon = ?, cityName = ? WHERE id = ?", [lat, lon, cityName, userId]); res.json({ success: true }); }); }); // 3. المراقبة التلقائية (تتم كل دقيقة) setInterval(async () => { db.all("SELECT * FROM users", async (err, users) => { if (err || !users) return; // جلب زلازل آخر ساعة فقط const startTime = new Date(Date.now() - 3600000).toISOString(); const quakeRes = await axios.get(`https://earthquake.usgs.gov/fdsnws/event/1/query?format=geojson&minmagnitude=4.0&starttime=${startTime}`).catch(() => null); if (!quakeRes || !quakeRes.data.features) return; for (let user of users) { if (user.lat === 0) continue; for (let quake of quakeRes.data.features) { const mag = quake.properties.mag; const dist = calculateDistance(user.lat, user.lon, quake.geometry.coordinates[1], quake.geometry.coordinates[0]); // منطق خطر ذكي: قوة عالية بمسافة كبيرة أو قوة متوسطة بمسافة قريبة جداً const isDangerous = (mag >= 5.0 && dist < 300) || (mag >= 4.0 && dist < 50); if (isDangerous && user.lastQuakeId !== quake.id) { await safeSendTelegram(user.chatId, `🚨 إنذار زلزال: ${mag} ريختر على بعد ${Math.round(dist)} كم.`); db.run("UPDATE users SET lastQuakeId = ? WHERE id = ?", [quake.id, user.id]); } } } }); }, 60000); // دالة حساب المسافة function calculateDistance(lat1, lon1, lat2, lon2) { const R = 6371; const dLat = (lat2 - lat1) * Math.PI / 180; const dLon = (lon2 - lon1) * Math.PI / 180; const a = Math.sin(dLat/2)**2 + Math.cos(lat1*Math.PI/180) * Math.cos(lat2*Math.PI/180) * Math.sin(dLon/2)**2; return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a)); } // 4. إرسال آمن (مع معالجة أخطاء) async function safeSendTelegram(chatId, text) { try { await axios.get(`https://api.telegram.org/bot${BOT_TOKEN}/sendMessage?chat_id=${chatId}&text=${encodeURIComponent(text)}`); } catch (e) { console.error("فشل إرسال التنبيه:", e.message); } } app.listen(3000, () => console.log("System Online 🚀"));
