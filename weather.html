const express = require('express'); const axios = require('axios'); const { Pool } = require('pg'); const crypto = require('crypto'); const app = express(); app.use(express.json()); const BOT_TOKEN = process.env.BOT_TOKEN; const TELEGRAM_SECRET = process.env.TELEGRAM_SECRET; if (!BOT_TOKEN) { console.error("❌ خطأ: BOT_TOKEN غير موجود في متغيرات البيئة!"); process.exit(1); } if (!TELEGRAM_SECRET) { console.error("❌ خطأ أمني قاتل: يجب تعيين متغير البيئة TELEGRAM_SECRET بشكل صارم!"); process.exit(1); } const pool = new Pool({ connectionString: process.env.DATABASE_URL, ssl: { rejectUnauthorized: false } }); // تهيئة الجداول المركزية (جدول المستخدمين + جدول منفصل للتنبيهات المرصودة لكل مستخدم) async function initDB() { await pool.query(` CREATE TABLE IF NOT EXISTS users ( id TEXT PRIMARY KEY, token TEXT, chatId TEXT UNIQUE, lat REAL, lon REAL, cityName TEXT ) `); await pool.query(` CREATE TABLE IF NOT EXISTS user_alerts ( user_id TEXT, quake_id TEXT, PRIMARY KEY (user_id, quake_id) ) `); } initDB().catch(err => console.error("Database Init Error:", err)); // 1. ويب هوك آمن تماماً للربط مع تيليجرام app.post('/api/telegram-webhook', async (req, res) => { const secretHeader = req.headers['x-telegram-bot-api-secret-token']; if (secretHeader !== TELEGRAM_SECRET) { return res.status(403).json({ error: "Unauthorized Webhook Request" }); } try { const update = req.body; if (update.message && update.message.text === '/start') { const chatId = update.message.chat.id; const userId = 'user_' + crypto.randomUUID().slice(0, 8); const token = crypto.randomBytes(16).toString('hex'); await pool.query( `INSERT INTO users (id, token, chatId, lat, lon, cityName) VALUES ($1, $2, $3, 0, 0, 'غير محدد') ON CONFLICT (chatId) DO NOTHING`, [userId, token, chatId] ); const result = await pool.query(`SELECT id, token FROM users WHERE chatId = $1`, [chatId]); if (result.rows.length > 0) { const user = result.rows[0]; await safeSendTelegram(chatId, `أهلاً بك!\nبيانات ربط موقعك بصفحة الويب:\n\nUserID: ${user.id}\nToken: ${user.token}`); } } } catch (e) { console.error("Webhook Error:", e.message); } res.sendStatus(200); }); // 2. تحديث الموقع مع التحقق الجغرافي الصارم app.post('/api/update-location', async (req, res) => { const { userId, token, lat, lon, cityName } = req.body; if (typeof lat !== 'number' || typeof lon !== 'number' || lat < -90 || lat > 90 || lon < -180 || lon > 180) { return res.status(400).json({ message: "إحداثيات جغرافية غير صالحة" }); } try { const userResult = await pool.query("SELECT * FROM users WHERE id = $1 AND token = $2", [userId, token]); if (userResult.rows.length === 0) return res.status(403).json({ message: "غير مصرح أو بيانات غير صحيحة" }); await pool.query( "UPDATE users SET lat = $1, lon = $2, cityName = $3 WHERE id = $4", [lat, lon, cityName || "موقعك الحالي", userId] ); res.json({ success: true }); } catch (e) { res.status(500).json({ message: "خطأ في السيرفر" }); } }); // 3. محرك المراقبة الدوري مع قفل استشاري (Advisory Lock) لضمان عدم تكرار الفحص بين الـ Instances setInterval(async () => { const client = await pool.connect(); try { // محاولة أخذ قفل فريد برقم معنوي (مثلاً 998877) لضمان أن نسخة واحدة فقط من الـ Server تفحص في هذه الدقيقة const lockResult = await client.query("SELECT pg_try_advisory_lock(998877)"); const acquired = lockResult.rows[0].pg_try_advisory_lock; if (!acquired) { // نسخة أخرى تقوم بالفحص حالياً، نتخطى هذه الدورة لتوفير استهلاك الـ API return; } const usersResult = await client.query("SELECT * FROM users"); const users = usersResult.rows; if (!users || users.length === 0) return; const startTime = new Date(Date.now() - 3600000).toISOString(); const quakeRes = await axios.get(`https://earthquake.usgs.gov/fdsnws/event/1/query?format=geojson&minmagnitude=4.0&starttime=${startTime}`).catch(() => null); const recentQuakes = quakeRes?.data?.features || []; for (let user of users) { if (user.lat === 0 && user.lon === 0) continue; for (let quake of recentQuakes) { const mag = quake.properties.mag; const quakeId = quake.id; const dist = calculateDistance(user.lat, user.lon, quake.geometry.coordinates[1], quake.geometry.coordinates[0]); const isMatch = (mag >= 5.0 && dist < 300) || (mag >= 4.0 && dist < 50); if (isMatch) { const insertCheck = await client.query( `INSERT INTO user_alerts (user_id, quake_id) VALUES ($1, $2) ON CONFLICT (user_id, quake_id) DO NOTHING`, [user.id, quakeId] ); if (insertCheck.rowCount > 0) { await safeSendTelegram(user.chatId, `⚠️ تنبيه زلزالي مرصود:\nالقوة: ${mag} ريختر\nالمسافة التقريبية: ${Math.round(dist)} كم عن (${user.cityName}).`); } } } } } catch (e) { console.error("خطأ في دورة المراقبة:", e.message); } finally { // تحرير القفل فور الانتهاء لتعمل الدورات القادمة بسلاسة await client.query("SELECT pg_advisory_unlock(998877)").catch(() => {}); client.release(); } }, 60000); function calculateDistance(lat1, lon1, lat2, lon2) { const R = 6371; const dLat = (lat2 - lat1) * Math.PI / 180; const dLon = (lon2 - lon1) * Math.PI / 180; const a = Math.sin(dLat/2)**2 + Math.cos(lat1*Math.PI/180) * Math.cos(lat2*Math.PI/180) * Math.sin(dLon/2)**2; return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a)); } async function safeSendTelegram(chatId, text) { try { await axios.get(`https://api.telegram.org/bot${BOT_TOKEN}/sendMessage`, { params: { chat_id: chatId, text: text } }); } catch (error) { console.error(`فشل إرسال تيليجرام للـ ID ${chatId}:`, error.message); } } const PORT = process.env.PORT || 3000; app.listen(PORT, () => console.log(`السيرفر يعمل بأعلى كفاءة مع قفل الـ Instances على البورت ${PORT} 🚀`));
